local function RunCommand(cmd)
    if not pcall(function() os.execute(cmd) end) then
        pcall(function()
            local p = io.popen(cmd)
            if p then
                p:read("a")
                p:close()
            end
        end)
    end
end

local function EachLine(path, fn)
    local f = io.open(path, "r")
    if f then
        for line in f:lines() do fn(line) end
        f:close()
    end
end

local function AppendLine(path, text)
    local f = io.open(path, "a")
    if not f then
        return false
    end
    f:write(text .. "\n")
    f:close()
    return true
end

local function WriteLines(path, lines)
    local f = io.open(path, "w")
    if not f then
        return false
    end
    for _, line in ipairs(lines) do f:write(line .. "\n") end
    f:close()
    return true
end

local function ReadLock()
    local owner, stamp
    EachLine(Files.lock, function(line)
        if not owner then
            owner = line
        elseif not stamp then
            stamp = tonumber(line)
        end
    end)
    return (owner ~= "" and owner or nil), stamp
end

local function AcquireLock(me)
    local limit = os.time() + 15
    while os.time() < limit do
        local owner, stamp = ReadLock()
        if owner == nil or stamp == nil or os.time() - stamp > 10 then
            WriteLines(Files.lock, { me, tostring(os.time()) })
            Sleep(100, 250)
            if ReadLock() == me then
                return true
            end
        end
        Sleep(150, 400)
    end
    return false
end

local MyNameCache = nil

local function GetMyName()
    if not MyNameCache then
        local name = Try(function() return getLocal().name end)
            or Try(function() return bot.name end)
        if type(name) == "string" and name ~= "" then
            MyNameCache = name
        end
    end
    return MyNameCache or "bot"
end

local function WithLock(fn)
    local me = GetMyName()
    if not AcquireLock(me) then
        return false
    end
    local ok, result = pcall(fn)
    if ReadLock() == me then
        WriteLines(Files.lock, {})
    end
    return ok and result
end

local function WaitCalm(seconds)
    local last, calm, limit = CurrentWorld(), 0, os.time() + 120
    while calm < seconds and os.time() < limit do
        WaitOnline()

        local now = CurrentWorld()

        else
            last, calm = now, 0
        end
    end

local function AddHistory(msg)
    AppendLine(Files.log, string.format("`%s` %s", os.date("%H:%M"), msg))
end

local function ReadHistory(n)
    local lines = {}
    EachLine(Files.log, function(line)
        if line ~= "" then
            table.insert(lines, line)
        end
    end)
    if #lines > 300 then
        local keep = {}
        for i = #lines - 29, #lines do table.insert(keep, lines[i]) end
        WriteLines(Files.log, keep)
    end
    local result = {}
    for i = #lines, math.max(1, #lines - n + 1), -1 do
        table.insert(result, lines[i])
    end
    return result
end

local function StatusText(st)
    if st == BotStatus.online then
        return "Online"
    end
    if st == BotStatus.offline then
        return "Offline"
    end
    return tostring(st)
end

local function UpdateStatus()
    pcall(function()
        local changed = false
        for idx, b in pairs(getBots() or {}) do
            local key, st = tostring(idx), b.status
            if PrevStatus[key] ~= st then
                if PrevStatus[key] ~= nil then
                    AddHistory(string.format(
                        "**Bot %s**: `%s` -> `%s`",
                        key, StatusText(PrevStatus[key]), StatusText(st)
                    ))
                end
                PrevStatus[key], changed = st, true
            end
        end
        if changed then
            local lines = {}
            for key, st in pairs(PrevStatus) do
                table.insert(lines, key .. "|" .. tostring(st))
            end
            WriteLines(Files.status, lines)
        end
    end)
end

local function ParseEntry(line)
    local clean = string.gsub(line, "%s+", "")
    local name, door = string.match(clean, "^([^:|]+)[:|]?(.*)$")
    if name and name ~= "" then
        return string.upper(name), door
    end
end

local function LoadWorldSet(path)
    local set = {}
    EachLine(path, function(line)
        local name, door = ParseEntry(line)
        if name then
            set[name] = door
        end
    end)
    return set
end

local function CountSet(set)
    local n = 0

    return n
end

local function escapeJson(str)
    if not str then
        return ""
    end
    str = tostring(str)
    str = str:gsub("\\", "\\\\")
    str = str:gsub('"', '\\"')
    str = str:gsub("\n", "\\n")
    str = str:gsub("\r", "\\r")
    -- karakter kontrol lain (tab dsb.) harus di-escape agar JSON valid
    str = str:gsub("%c", function(c) return string.format("\\u%04x", string.byte(c)) end)
    -- buang byte non-ASCII yang rusak (penyebab utama Discord menolak dengan 400)
    str = str:gsub("[\128-\255]", "")
    return str
end

local function kirimWebhook(metode, url, payload)
    local status, body = nil, nil
    local ok, err = pcall(function()
        local client = HttpClient.new()
        client.url = url
        client.method = metode
        client.content = payload
        client.headers["Content-Type"] = "application/json; charset=utf-8"
        local hasil = client:request()
        if hasil then
            status = hasil.status
            body = hasil.body
        end
    end)
    if not ok then
        print("[WEBHOOK] Error HTTP: " .. tostring(err))
    end
    return status, body
end

local function ReadHookStamp()

-- Mengembalikan true jika berhasil, atau false + kode status jika gagal.
local function DeliverWebhook(payload)
    local status

    if WebhookMessageID ~= "" then
        status = kirimWebhook(Method.patch, WebhookURL .. "/messages/" .. WebhookMessageID, payload)
        if status == 200 then
            return true
        end
        if status ~= 404 then
            return false, status
        end
        WebhookMessageID = "" -- pesan lama sudah tidak ada, buat pesan baru
    end

    local body
    status, body = kirimWebhook(Method.post, WebhookURL .. "?wait=true", payload)
    if status == 200 or status == 204 then
        -- simpan ID pesan supaya laporan berikutnya meng-edit pesan yang sama
        local id = type(body) == "string" and string.match(body, '"id"%s*:%s*"(%d+)"')
        if id then
            WebhookMessageID = id
            WriteLines(Files.msgid, { id })
        end
        return true
    end
    return false, status
end

        end
    end
    return text
end

-- Semua bot boleh mengirim, tapi dibatasi lewat file bersama (dfarm_hook.txt)
-- agar tidak double. Tidak lagi bergantung pada "leader" yang bisa saja macet.
local function CheckWebhook()
    if WebhookURL == "" or WebhookBusy then
        return
    end

    -- jika sering gagal, perlambat otomatis (maks 6x) agar tidak kena rate-limit
    local interval = IntervalWebhook * (1 + math.min(WebhookFailCount, 5))
    if os.time() - LastWebhookTime < interval then
        return
    end

    LastWebhookTime = os.time()
    if os.time() - ReadHookStamp() < interval - 1 then
        return
    end

    SendWebhook()
end

    return true
end

end

    end

    local cands = {}
    for _, o in ipairs(Offsets) do
        local sx, sy = x + o[1], y + o[2]
        if InBounds(sx, sy) then
            local fg = GetTile(sx, sy)
            if fg <= 0 or Passable[fg] then
                table.insert(cands, { sx, sy })
            end
        end
    end

    for _, c in ipairs(cands) do
        pcall(function() bot:findPath(c[1], c[2]) end)
        Sleep(120, 150)
        if Near(x, y) and not OnTile(x, y) then
            return true
        end
    end
    return false
end

local function StandAt(sx, sy)
    for _ = 1, 3 do
        if OnTile(sx, sy) then
            return true
        end
        pcall(function() bot:findPath(sx, sy) end)
        Sleep(300, 500)
    end
    return OnTile(sx, sy)
end

-- Baca angka dari salah satu nama properti yang ada pada objek.
local function ReadNumber(obj, names)
    for _, name in ipairs(names) do
        local value = Try(function() return obj[name] end)
        if type(value) == "number" then
            return value
        end
    end
end

-- Nama properti yang dicoba untuk UID pemilik pada data extra lock.
local OwnerFields = {
    "owner", "owner_uid", "ownerUid", "owner_id", "ownerid", "ownerID",
    "uid", "user_id", "userid",
}

-- Data extra sebuah tile (di sini tersimpan pemilik lock), nil jika tidak ada.
local function TileExtraOf(x, y)
    local tile = getTile(x, y)
    if not tile then
        return nil
    end
    return Try(function()
        if tile:hasExtra() then
            return tile:getExtra()
        end
    end)
end

-- UID pemilik lock di tile (x, y), nil jika tidak terbaca.
local function TileOwnerUid(x, y)
    local extra = TileExtraOf(x, y)
    if not extra then
        return nil
    end
    return ReadNumber(extra, OwnerFields)
end

-- UID akun bot ini (getLocal().userid), nil jika tidak terbaca. UID asli
-- berupa angka besar, angka kecil diabaikan supaya tidak salah dibandingkan.
local function MyUid()
    local uid = Try(function() return getLocal().userid end)
    if type(uid) == "number" and uid > 100000 then
        return uid
    end
end

-- true/false jika engine menjawab, nil jika jawabannya tidak jelas.
local function BotHasAccess(x, y)
    local result = Try(function() return getWorld():hasAccess(x, y) end)
    if type(result) == "boolean" then
        return result
    end
end

-- Tampilkan isi data extra lock ke console agar nama propertinya bisa dicocokkan.
local function DumpTile(x, y)
    local extra = TileExtraOf(x, y)
    if not extra then
        print(string.format("[LOCK] tile (%d,%d) tidak punya data extra", x, y))
        return
    end

    local parts = {}
    pcall(function()
        for key, value in pairs(extra) do
            table.insert(parts, tostring(key) .. "=" .. tostring(value))
        end
    end)
    print("[LOCK] extra (pairs): " .. table.concat(parts, ", "))

    local found = {}
    local common = {
        "owner", "owner_uid", "ownerUid", "owner_id", "ownerid", "uid",
        "user_id", "userid", "admin", "admins", "access", "access_list",
        "accessList", "flags", "label", "name", "type",
    }
    for _, name in ipairs(common) do
        local value = Try(function() return extra[name] end)
        if value ~= nil then
            table.insert(found, name .. "=" .. tostring(value))
        end
    end
    print("[LOCK] extra (nama umum): " .. table.concat(found, ", "))
end

-- true hanya jika bot terbukti TIDAK punya akses ke lock ini:
--   * engine menyatakan bot punya akses  -> bukan milik orang lain
--   * UID pemilik dan UID bot terbaca dan berbeda -> milik orang lain
--   * selain itu (data tidak jelas) -> dianggap milik bot (perilaku aman)
local function LockOwnedByOther(x, y)
    local owner, me, access = TileOwnerUid(x, y), MyUid(), BotHasAccess(x, y)
    print(string.format("[LOCK] (%d,%d) id=%d owner=%s bot=%s akses=%s",
        x, y, GetTile(x, y), tostring(owner), tostring(me), tostring(access)))

    if not owner then
        DumpTile(x, y)
    end

    if access == true then
        return false
    end
    if owner and me then
        return owner ~= me
    end
    return false
end

-- Berdiri di (sx, sy) jika diberikan, supaya jarak pukul pasti dekat.
-- Jika tidak, bot cukup mendekat ke area tile (perilaku lama).
local function Reach(x, y, sx, sy)
    if sx then
        return StandAt(sx, sy)
    end
    return Approach(x, y)
end

-- Ada batas waktu dan batas pukulan, sehingga lock yang tidak bisa dihancurkan
-- tidak membuat bot menunggu selamanya. Jika lock belum hancur setelah
-- MaxHitLockAsing pukulan dan pemiliknya terbukti orang lain: ForeignLock = true.
local function BreakTile(x, y, sx, sy)
    if not InBounds(x, y) then
        return false
    end

    local persistent = LockIDs[GetTile(x, y)] == true
    local hits, ownerChecked = 0, false
    local deadline = os.time() + TimeoutBreakTile

    local function Fail(reason)
        print(string.format("[BREAK] (%d,%d) id=%d gagal: %s (pukulan=%d)",
            x, y, GetTile(x, y), reason, hits))
        return false
    end

    while os.time() < deadline do
        if not StillHere() then
            return Fail("bot pindah world")
        end

        local fg = GetTile(x, y)
        if fg <= 0 then
            return true
        end
        if Unbreakable[fg] then
            return Fail("blok tidak bisa dihancurkan")
        end
        if not persistent and hits >= 40 then
            return Fail("tidak hancur setelah 40 pukulan")
        end

        -- Lock belum hancur setelah banyak pukulan: cek pemiliknya (sekali saja)
        if persistent and hits >= MaxHitLockAsing and not ownerChecked then
            ownerChecked = true
            if LockOwnedByOther(x, y) then
                ForeignLock = true
                return Fail("lock milik orang lain")
            end
        end
        if persistent and hits >= MaxHitLock then
            return Fail("lock tidak hancur setelah " .. MaxHitLock .. " pukulan")
        end

        WaitOnline()
        if Reach(x, y, sx, sy) then
            bot:hit(x, y)
            hits = hits + 1
            -- Pakai sleep biasa (tanpa cek webhook): request webhook yang
            -- memblokir bisa membuat damage blok pulih sebelum sempat pecah.
            sleep(math.random(120, 140))
        else
            Sleep(1000, 2000)
        end
    end
    return Fail("waktu habis")
end

local function PlaceAt(x, y, id)
    if not InBounds(x, y) or not StillHere() or GetTile(x, y) ~= 0 or Count(id) < 1 then
        return false
    end
    WaitOnline()
    if not Approach(x, y) then
        return false
    end
    bot:place(x, y, id)
    Sleep(200, 300)
    return GetTile(x, y) == id
end

local function FindDoor()
    for y = 0, 59 do
        for x = 0, 99 do
            if GetTile(x, y) == ID_MAINDOOR then
                return x, y
            end
        end
    end
end

local function LockTiles()
    local list = {}
    for y = 0, 59 do
        for x = 0, 99 do
            if LockIDs[GetTile(x, y)] then
                table.insert(list, { x, y })
            end
        end
    end
    return list
end

-- Setelah warp, data tile kadang belum selesai dimuat sehingga (1,2) terbaca
-- kosong padahal ada lock. Tunggu sampai data world terbaca.
local function WaitTileReady()
    for _ = 1, 10 do
        if CurrentWorld() ~= "" and getTile(1, 2) ~= nil then
            break
        end
        Sleep(500, 700)
    end
    Sleep(800, 1200)
end

-- Hancurkan tile (1,2) dari posisi (0,2). Diulang beberapa kali karena
-- kadang data tile terlambat terbaca atau blok sempat pulih.
local function BreakFrontTile()
    for _ = 1, 3 do
        StandAt(0, 2)
        local fg = GetTile(1, 2)
        if fg <= 0 or fg == FillId then
            return true
        end
        if BreakTile(1, 2, 0, 2) then
            return true
        end
        if ForeignLock or not StillHere() then
            return false
        end
        Sleep(800, 1200)
    end
    print(string.format("[BREAK] tile (1,2) masih terisi id %d", GetTile(1, 2)))
    return false
end

-- Pasang blok pengisi (Platform/Dirt) di (1,2) setelah dibersihkan.
local function FillFrontTile()
    for _ = 1, 5 do
        if GetTile(1, 2) ~= 0 then
            break
        end
        StandAt(0, 2)
        PlaceAt(1, 2, FillId)
        Sleep(300, 500)
    end
end

local function ClearDoorArea(dx, dy)
    WaitTileReady()
    if dx and LockIDs[GetTile(dx, dy - 1)] then
        BreakTile(dx, dy - 1)
    end
    BreakTile(0, 2)
    BreakFrontTile()
    FillFrontTile()
end

local function BreakSmallLock()
    WaitTileReady()
    BreakFrontTile()
    FillFrontTile()
end

-- ============================================================
-- VERIFIKASI & PEMBERSIHAN WORLD
-- ============================================================

local function VerifyWorld(target)
    if CurrentWorld() ~= target then
        return false, "bot tidak berada di world target"
    end

    if Mode == "ROTASI" then
        local fg = GetTile(1, 2)
        -- Ubah verifikasi agar mengecek segala jenis lock, bukan hanya SL
        if LockIDs[fg] then
            return false, "lock di (1,2) belum hancur"
        end
        if fg <= 0 then
            return false, string.format("(1,2) belum terisi platform (tile=%d)", fg)
        end
        return true, "ok"
    end

    local locks = LockTiles()
    if #locks > 0 then
        return false, string.format("masih ada lock di (%d,%d)", locks[1][1], locks[1][2])
    end

    local fgPlat = GetTile(1, 2)
    if fgPlat <= 0 then
        return false, string.format("(1,2) belum terisi platform (tile=%d)", fgPlat)
    end

    return true, "ok"
end

-- Return: (berhasil, terputus, lockOrangLain)
-- Maksimal MaxAttemptBersih percobaan, lalu dianggap gagal agar counter
-- MaxGagal/JedaUlang bekerja. Jika ketemu lock orang lain, langsung berhenti.
local function CleanWithVerify(target, door)
    local warpStr = (door and door ~= "") and (target .. "|" .. door) or target
    local attempt, warpFails = 0, 0

    while true do
        attempt = attempt + 1
        WaitOnline()
        HandOverToMalady(nil)
        ExpectedWorld = ""
        ForeignLock = false

        local inWorld = CurrentWorld() == target
            or WarpTo(warpStr)
            or (warpStr ~= target and WarpTo(target))

        if not inWorld then
            warpFails = warpFails + 1
            if warpFails >= 3 then
                return false, false
            end
        else
            warpFails, ExpectedWorld = 0, target

            if Mode == "ROTASI" then
                BreakSmallLock()
            else
                local dx, dy = FindDoor()
                ClearDoorArea(dx, dy)
                for _, p in ipairs(LockTiles()) do
                    BreakTile(p[1], p[2])
                    if ForeignLock then
                        break
                    end
                end
            end

            if ForeignLock then
                ExpectedWorld = ""
                return false, false, true
            end

            CleanJunk()

            if StillHere() then
                local ok1 = VerifyWorld(target)
                if ok1 then
                    Sleep(2000, 3000)
                    local ok2 = VerifyWorld(target)
                    if ok2 then
                        ExpectedWorld = ""
                        return true, false
                    end
                end
            end

            if not StillHere() then
                ExpectedWorld = ""
                return false, true
            end
        end

        if attempt >= MaxAttemptBersih then
            ExpectedWorld = ""
            return false, false
        end
        Sleep(1500, 2500)
    end
end

local function MissingItem()
    for id, quota in pairs(ItemKuota) do
        if Count(id) < quota[1] then
            return id
        end
    end
end

-- Jumlah objek (item jatuh) per tile, kunci = "x,y". Kosong jika tidak terbaca.
local function ObjectCounts()
    local counts = {}

    local function Add(obj)
        local x = Try(function() return obj.x end)
        local y = Try(function() return obj.y end)
        if type(x) == "number" and type(y) == "number" then
            local key = math.floor(x / 32) .. "," .. math.floor(y / 32)
            counts[key] = (counts[key] or 0) + 1
        end
    end

    local list = Try(function() return getWorld():getObjects() end)
    if list == nil then
        return counts
    end

    local ok = pcall(function()
        for _, obj in pairs(list) do Add(obj) end
    end)
    if not ok or next(counts) == nil then
        pcall(function()
            for i = 0, list:size() - 1 do Add(list:get(i)) end
        end)
    end
    return counts
end

-- Tile berikutnya untuk drop: +1 ke kanan dari posisi bot (baris yang sama).
-- Tile yang tidak bisa dipijak atau sudah penuh dilewati, terus ke kanan.
-- Jika sampai ujung kanan tidak ada yang cocok, cari tile terdekat di mana saja.
local function FindDropSpot(tried)
    local bx, by = Pos()
    local counts = ObjectCounts()

    for x = bx + 1, 98 do
        local key = x .. "," .. by
        local fg, below = GetTile(x, by), GetTile(x, by + 1)
        if not tried[key] and fg <= 0 and below > 0 and not Passable[below]
            and (counts[key] or 0) < MaxObjekPerTile then
            return { x, by }
        end
    end

    local best, bestDist
    for y = 1, 58 do
        for x = 1, 98 do
            local key = x .. "," .. y
            local fg, below = GetTile(x, y), GetTile(x, y + 1)
            if not tried[key] and fg <= 0 and below > 0 and not Passable[below]
                and (counts[key] or 0) < MaxObjekPerTile then
                local dist = math.abs(x - bx) + math.abs(y - by) * 2
                if not bestDist or dist < bestDist then
                    best, bestDist = { x, y }, dist
                end
            end
        end
    end
    return best
end

-- Drop item `id` sampai sisa di tas tinggal `keep`. Jika drop tidak mengurangi
-- jumlah item (tile penuh), bot pindah 1 tile ke kanan lalu mencoba lagi.
local function DropItem(id, keep)
    local tried, moves = {}, 0
    local bx, by = Pos()
    tried[bx .. "," .. by] = true

    while Count(id) > keep do
        WaitOnline()
        local before = Count(id)
        pcall(function() bot:drop(id, before - keep) end)
        Sleep(800, 1200)

        if Count(id) >= before then
            pcall(function() bot:fastDrop(id, before - keep) end)
            Sleep(800, 1200)
        end

        if Count(id) >= before then
            if moves >= MaxPindahDrop then
                print(string.format("[DROP] item %d gagal didrop setelah %d kali pindah", id, moves))
                return false
            end

            local spot = FindDropSpot(tried)
            if not spot then
                print("[DROP] tidak ada tile kosong untuk drop item " .. id)
                return false
            end

            moves = moves + 1
            tried[spot[1] .. "," .. spot[2]] = true
            StandAt(spot[1], spot[2])
            Sleep(300, 500)
        end
    end
    return Count(id) <= keep
end

local function DropSurplus(storageName)
    if CurrentWorld() ~= storageName then
        return
    end
    for id, quota in pairs(ItemKuota) do
        if Count(id) > quota[2] then
            DropItem(id, quota[2])
        end
    end
end

local function CollectAndVerify(storageName)
    for _ = 1, MaxRoundCollect do
        local missing = MissingItem()
        if missing then
            pcall(function() bot.auto_collect = true end)
            Sleep(1000)
            pcall(function() bot.auto_collect = false end)
            Sleep(500, 800)
        end

        DropSurplus(storageName)

        local stable = true
        for _ = 1, VerifyDetik do
            Sleep(1000)
            if MissingItem() then
                stable = false
                break
            end
        end
        if stable then
            return true
        end
    end
    return false
end

local function RestockAtStorage(storageName)
    if not WarpTo((string.gsub(WorldStorage, ":", "|"))) then
        WarpTo(storageName)
    end
    Sleep(1000, 2000)
    return CollectAndVerify(storageName)
end

-- ============================================================
-- PICKAXE (ambil -> pakai -> kembali ke home -> leveling)
-- ============================================================

local function WorldNameOf(entry)
    return string.upper(string.match(entry, "^([^|:]+)") or entry)
end

-- ID item yang sedang dipakai di tangan, nil jika tidak terbaca.
local function WornHandItem()
    local clothes = Try(function() return getLocal().clothes end)
    if not clothes then
        return nil
    end

    local names = { "hand", "hand_item", "handheld", "tool" }
    for _, name in ipairs(names) do
        local value = Try(function() return clothes[name] end)
        if type(value) == "number" then
            return value
        end
    end
end

-- Pakai Pickaxe. Memakai item yang sudah terpasang justru melepasnya, jadi:
--   * slot tangan terbaca  -> pakai hanya jika belum terpasang
--   * slot tangan tidak terbaca -> pakai hanya jika Pickaxe baru saja diambil
local function WearPickaxe(justCollected)
    if Count(PickaxeId) < 1 then
        return
    end

    local worn = WornHandItem()
    if worn == PickaxeId then
        return
    end
    if worn == nil and not justCollected then
        return
    end

    pcall(function() bot:wear(PickaxeId) end)
    Sleep(500, 800)
end

-- Nama world home: dari hasil SetupHomeWorld, atau dari pengaturan bot.home_world.
local function HomeWorldName(levelWorld)
    if levelWorld and levelWorld ~= "" then
        return levelWorld
    end

    local home = Try(function() return bot.home_world end)
    if type(home) == "string" and home ~= "" then
        return string.upper(home)
    end
end

-- Kembali ke home. Jika nama home tidak diketahui, pakai perintah /home.
local function GoHome(homeName)
    if homeName then
        if CurrentWorld() == homeName then
            return true
        end
        return WarpTo(homeName)
    end

    pcall(function() bot:say("/home") end)
    Sleep(5000, 6000)
    return CurrentWorld() ~= ""
end

-- Masuk ke PickaxeStorage lalu kumpulkan Pickaxe. Hanya 1 yang disimpan,
-- sisanya di-drop lagi di world itu.
local function CollectPickaxe()
    local storageName = WorldNameOf(PickaxeStorage)
    if not WarpTo((string.gsub(PickaxeStorage, ":", "|"))) then
        WarpTo(storageName)
    end
    if CurrentWorld() ~= storageName then
        return false
    end
    Sleep(1000, 2000)

    for _ = 1, MaxRoundCollect do
        if Count(PickaxeId) >= 1 then
            break
        end

        pcall(function() bot.auto_collect = true end)
        Sleep(1500, 2000)
        pcall(function() bot.auto_collect = false end)
        Sleep(500, 800)
    end

    if Count(PickaxeId) > 1 then
        DropItem(PickaxeId, 1)
    end
    return Count(PickaxeId) >= 1
end

-- Urutan: ambil Pickaxe (jika belum punya) -> pakai -> kembali ke home.
-- Mengembalikan nama world untuk leveling (home jika perjalanan dilakukan).
local function PickaxeTrip(levelWorld)
    if not AutoWearPickaxe then
        return levelWorld
    end

    local needTrip = Count(PickaxeId) < 1 and PickaxeStorage ~= ""
    if not needTrip then
        WearPickaxe(false)
        return levelWorld
    end

    local home = HomeWorldName(levelWorld)

    if CollectPickaxe() then
        WearPickaxe(true)
        AddHistory(string.format("Bot %s: Pickaxe berhasil diambil", tostring(GetMyIndex())))
    else
        AddHistory(string.format("Bot %s: Pickaxe tidak ditemukan di PickaxeStorage",
            tostring(GetMyIndex())))
    end

    GoHome(home)
    return home or levelWorld
end

-- ============================================================
-- SEED STORAGE (harvest otomatis + simpan ke world seed)
-- ============================================================

local SeedDone = {} -- world yang sudah diharvest
local SeedRetryAt = 0 -- jeda sebelum mencoba lagi jika penyimpanan gagal

-- Pemicu: ada blok Dirt di tile SeedTrigger pada world saat ini.
local function SeedTriggerActive()
    if SeedStorage == "" or os.time() < SeedRetryAt then
        return false
    end

    local world = CurrentWorld()
    if world == "" or world == "EXIT" or SeedDone[world] then
        return false
    end
    if world == WorldNameOf(SeedStorage) or world == WorldNameOf(WorldStorage) then
        return false
    end

    return GetTile(SeedTrigger[1], SeedTrigger[2]) == ID_DIRT
end

-- Pohon (seed yang tertanam) selalu punya ID ganjil.
local function IsTree(fg)
    return fg > 1 and fg % 2 == 1
end

-- Daftar pohon, diurutkan zig-zag per baris supaya jalan bot pendek.
local function TreeList()
    local list, leftToRight = {}, true
    for y = 0, 59 do
        local first, last, step = 0, 99, 1
        if not leftToRight then
            first, last, step = 99, 0, -1
        end

        local found = false
        for x = first, last, step do
            if IsTree(GetTile(x, y)) then
                table.insert(list, { x, y })
                found = true
            end
        end
        if found then
            leftToRight = not leftToRight
        end
    end
    return list
end

local function HarvestTile(x, y)
    for _ = 1, 20 do
        if not IsTree(GetTile(x, y)) or not StillHere() then
            return
        end
        if not Approach(x, y) then
            return
        end
        bot:hit(x, y)
        Sleep(120, 140)
    end
end

-- Pecahkan semua pohon sambil auto collect menangkap drop-nya.
local function HarvestWorld()
    local deadline = os.time() + TimeoutHarvest
    local trees = TreeList()

    for _, t in ipairs(trees) do
        if os.time() > deadline or not StillHere() then
            break
        end
        WaitOnline()
        HandOverToMalady(nil)
        pcall(function() bot.auto_collect = true end)
        HarvestTile(t[1], t[2])
    end

    Sleep(3000, 4000) -- beri waktu mengumpulkan drop terakhir
    pcall(function() bot.auto_collect = false end)
    return #trees
end

-- Buang semua item SeedItems ke SeedStorage. Item yang masuk kuota tas
-- (mis. Dirt) hanya dibuang kelebihannya.
local function StoreSeeds()
    local seedName = WorldNameOf(SeedStorage)
    if not WarpTo((string.gsub(SeedStorage, ":", "|"))) then
        WarpTo(seedName)
    end
    if CurrentWorld() ~= seedName then
        return false
    end
    Sleep(1000, 2000)

    pcall(function() bot.auto_collect = false end)
    local stored = {}
    for _, id in ipairs(SeedItems) do
        local keep = ItemKuota[id] and ItemKuota[id][2] or 0
        local before = Count(id)
        local extra = before - keep
        if extra > 0 then
            DropItem(id, keep)
            stored[id] = before - Count(id)
        end
    end

    AddSeedTotals(stored)
    return true
end

-- Urutan: auto build mati -> harvest -> simpan ke SeedStorage -> auto build nyala.
local function RunSeedHarvest(build)
    local world = CurrentWorld()
    SeedDone[world] = true
    ExpectedWorld = world

    SetBuild(build, false)
    AddHistory(string.format("Bot %s: harvest dimulai", tostring(GetMyIndex())))

    local startedAt = os.time()
    local trees = HarvestWorld()
    ExpectedWorld = ""
    print(string.format("[HARVEST] %s: %d pohon dalam %d detik",
        world, trees, os.time() - startedAt))

    if StoreSeeds() then
        AddHistory(string.format("Bot %s: %d pohon diharvest, seed disimpan",
            tostring(GetMyIndex()), trees))
    else
        SeedDone[world] = nil
        SeedRetryAt = os.time() + 60
        AddHistory(string.format("Bot %s: gagal masuk SeedStorage, coba lagi 60 detik",
            tostring(GetMyIndex())))
    end

    SetBuild(build, not BuildBlocked)
end

-- ============================================================
-- LEVELING
-- ============================================================

local function IsProtected(fg)
    return fg > 0 and (Unbreakable[fg] or LockIDs[fg]) and true or false
end

local function SetupHomeWorld()
    local hasFirst = false
    for _, p in ipairs(LockTiles()) do
        if GetTile(p[1], p[2]) == ID_FIRSTLOCK then
            hasFirst = true
        end
    end
    if not hasFirst then
        return nil
    end

    local name = CurrentWorld()
    pcall(function() bot.home_world = name end)
    return name
end

local function HasWork(x, y)
    local t = getTile(x, y)
    if not t then
        return false
    end
    if t.fg > 0 then
        return not IsProtected(t.fg)
    end
    return t.bg > 0
end

local function ClearForXP(x, y)
    for _ = 1, 60 do
        if not HasWork(x, y) or not Approach(x, y) then
            return
        end
        bot:hit(x, y)
        Sleep(115, 135)
    end
end

-- PERBAIKAN: ada batas waktu (TimeoutLeveling) agar bot tidak terjebak
-- selamanya jika level tidak naik / blok leveling habis.
local function RunLeveling(levelWorld)
    pcall(function() bot.auto_collect = true end)

    local tiles = 0
    local deadline = os.time() + TimeoutLeveling

    local function Finished()
        return GetLevel() >= TargetLevel or os.time() > deadline
    end

    local function Housekeep()
        WaitOnline()
        HandOverToMalady(nil)
        if tiles % 20 == 0 then
            EnableMalady()
            CleanJunk()
        end
        if levelWorld ~= nil and CurrentWorld() ~= levelWorld then
            pcall(function() bot.auto_collect = false end)
            WarpTo(levelWorld)
            pcall(function() bot.auto_collect = true end)
        end
    end

    local rightToLeft, done = true, false
    for y = 0, 59 do
        if done then
            break
        end

        local hasWork = false
        for x = 0, 99 do
            if HasWork(x, y) then
                hasWork = true
                break
            end
        end

        if hasWork then
            local first, last, step = 99, 0, -1
            if not rightToLeft then
                first, last, step = 0, 99, 1
            end

            for x = first, last, step do
                tiles = tiles + 1
                if Finished() then
                    done = true
                    break
                end
                Housekeep()
                if HasWork(x, y) then
                    ClearForXP(x, y)
                end
            end
            rightToLeft = not rightToLeft
        end
    end

    while not Finished() do
        tiles = tiles + 1
        Housekeep()

        local bx, by = Pos()
        local tx = IsProtected(GetTile(bx + 1, by)) and bx - 1 or bx + 1
        if InBounds(tx, by) and not IsProtected(GetTile(tx, by)) and Count(BlokLeveling) > 0 then
            if GetTile(tx, by) == 0 then
                bot:place(tx, by, BlokLeveling)
                Sleep(85, 105)
            end
            if GetTile(tx, by) > 0 then
                bot:hit(tx, by)
            end
        end
        Sleep(115, 135)
    end

    if GetLevel() < TargetLevel then
        AddHistory("Leveling timeout, lanjut tanpa mencapai target level")
    end
    pcall(function() bot.auto_collect = false end)
end

local function ApplyDoorId(build)
    -- Jika IdDoor masih "0", biarkan kosong/acak dari executor
    if IdDoor == "0" then return end 
    
    -- Langsung tembak ke properti executor yang sudah terkonfirmasi
    pcall(function() build.custom_id = tostring(IdDoor) end)
end

local function ApplyBuildConfig(build)
    local settings = {
        { "enabled", false }, { "soil_mode", false }, { "insert_rotation", true },
        { "ignore_lava", false }, { "ignore_cave", false }, { "build_water", false },
        { "build_plat", UsePlatform },
        { "use_fire_house", false },
        { "item_storage", WorldStorage },
        { "soil_storage", "" }, { "output_path", FileHasil }, { "name_length", PanjangNama },
        { "door_id", 12 }, { "entrance_id", 224 }, { "platform_id", 102 },
        -- MENAMBAHKAN FITUR PENCARIAN WORLD TANPA ANGKA
        { "no_number", true },
    }
    for _, s in ipairs(settings) do
        pcall(function() build[s[1]] = s[2] end)
    end
end

-- ============================================================
-- PROGRAM UTAMA
-- ============================================================

local function Main()
    SetupInternalFiles()
    if Mode ~= "PUTWL" and Mode ~= "ROTASI" then
        Mode = "PUTWL"
    end

    PrevStatus = LoadStatusFile()
    LoadMessageId()
    print("[START] Auto DirtFarm berjalan.")

    WaitOnline()
    Sleep(math.min(tonumber(GetMyIndex()) or 1, 30) * 300)

    pcall(function()
        bot.anti_fire = true
        bot.anti_toxic = true
        bot.auto_trash = false
    end)

    local levelWorld = SetupHomeWorld()
    EnableMalady()
    HandOverToMalady(nil)

    -- Ambil dan pakai Pickaxe dulu, kembali ke home, baru leveling
    levelWorld = PickaxeTrip(levelWorld)

    if GetLevel() < TargetLevel then
        RunLeveling(levelWorld)
    end

    local build = bot.auto_build
    if not build then
        print("[ERROR] bot.auto_build tidak tersedia, cek versi Lucifer / pengaturan bot.")
        return
    end

    ApplyBuildConfig(build)
    ApplyDoorId(build)
    pcall(function() bot.auto_collect = false end)
    SetBuild(build, false)

    local me = GetMyName()
    local storageName = string.upper(string.match(WorldStorage, "^([^|:]+)") or WorldStorage)
    local cleaned = LoadWorldSet(FileBersih)
    local owners = LoadOwners()
    local lastRefresh, lastRestock = os.time(), 0
    local myWorlds, fails, retryAt, cycles = {}, {}, {}, {}
    local LastIdleLog = 0
    local dropped = LoadWorldSet(FileGagal)

    for world, owner in pairs(owners) do
        if owner == me and not cleaned[world] and dropped[world] == nil then
            myWorlds[world] = true
        end
    end

    -- World penyimpanan (item, seed, pickaxe) bukan DirtFarm dan tidak dicatat
    local skipWorlds = { [storageName] = true }
    if SeedStorage ~= "" then
        skipWorlds[WorldNameOf(SeedStorage)] = true
    end
    if PickaxeStorage ~= "" then
        skipWorlds[WorldNameOf(PickaxeStorage)] = true
    end

    local function MarkCurrentWorld()
        local cur = CurrentWorld()
        if cur ~= "" and cur ~= "EXIT" and not skipWorlds[cur]
            and not cleaned[cur] and not myWorlds[cur] and dropped[cur] == nil then
            myWorlds[cur], owners[cur] = true, me
            AppendOwner(cur, me)
        end
    end

    SweepDirtfarm()

    local function Step()
        WaitOnline()
        EnableMalady()
        HandOverToMalady(build)
        CheckWebhook()
        MarkCurrentWorld()

        if SeedTriggerActive() then
            RunSeedHarvest(build)
            return
        end

        if os.time() - lastRefresh >= RefreshBersihDtk then
            cleaned, owners, lastRefresh = LoadWorldSet(FileBersih), LoadOwners(), os.time()
            if IsLeader() then
                SweepDirtfarm()
            end
        end

        local recorded = LoadWorldSet(FileHasil)

        -- PERBAIKAN: world yang sedang masa tunggu (cooldown) tidak lagi
        -- menahan auto build, sehingga bot tidak diam 5 menit penuh.
        local pendingMine = 0
        for w in pairs(myWorlds) do
            if recorded[w] and not cleaned[w] and (retryAt[w] or 0) <= os.time() then
                pendingMine = pendingMine + 1
            end
        end

        BuildBlocked = pendingMine > 0
        if BuildBlocked and BuildState ~= false then
            SetBuild(build, false)
        elseif not BuildBlocked and BuildState ~= true then
            SetBuild(build, true)
        end

        local target = nil
        for w in pairs(myWorlds) do
            if recorded[w] and not cleaned[w] and (retryAt[w] or 0) <= os.time() then
                target = w
                break
            end
        end

        if not target then
            if os.time() - LastIdleLog >= 60 then
                LastIdleLog = os.time()
                print(string.format(
                    "[STEP] tidak ada target (world=%s, milik bot=%d, tercatat di Proses.txt=%d)",
                    CurrentWorld(), CountSet(myWorlds), CountSet(recorded)
                ))
            end
            for _ = 1, 4 do
                Sleep(400, 600)
                MarkCurrentWorld()
            end
            return
        end

        SetBuild(build, false)
        Sleep(1500, 2000)

        -- Blok pengisi (1,2) habis: restock paksa, abaikan jeda 120 detik
        local fillEmpty = Count(FillId) < 1
        if fillEmpty or (MissingItem() and os.time() - lastRestock > 120) then
            lastRestock = os.time()
            RestockAtStorage(storageName)
        end

        -- Masih kosong setelah restock (stok storage habis): jangan buang percobaan,
        -- tunda world ini, kirim peringatan, dan coba lagi setelah jeda.
        if Count(FillId) < 1 then
            retryAt[target] = os.time() + JedaUlangDetik
            AddHistory(string.format("Bot %s: blok pengisi (id %d) habis, world ditunda",
                tostring(GetMyIndex()), FillId))
            SendWebhook("Blok pengisi (1,2) habis di tas dan storage")
            return
        end

        print("[STEP] membersihkan " .. target)
        local ok, interrupted, foreign = CleanWithVerify(target, recorded[target])
        print(string.format("[STEP] hasil %s: ok=%s terputus=%s lock_asing=%s",
            target, tostring(ok), tostring(interrupted), tostring(foreign)))

        if interrupted then
            if not HandOverToMalady(build) then
                WaitCalm(10)
            end
            return
        end

        if foreign then
            -- Lock milik orang lain: tidak bisa dibersihkan, buang dari daftar
            dropped[target] = ""
            myWorlds[target], fails[target], cycles[target] = nil, nil, nil
            DropWorld(target, "lock milik orang lain")
            AddHistory(string.format("Bot %s: world di-lock orang lain, dilewati",
                tostring(GetMyIndex())))
        elseif ok then
            cleaned[target], myWorlds[target], fails[target] = true, nil, nil
            
            -- Menyesuaikan format teks yang akan disimpan
            local teksSimpan = target
            if Mode == "ROTASI" then
                local doorTarget = recorded[target]
                if doorTarget and doorTarget ~= "" then
                    teksSimpan = target .. ":" .. doorTarget
                else
                    teksSimpan = target .. ":" .. IdDoor
                end
            end
            
            local saved = AppendLine(FileBersih, teksSimpan)
            print(string.format("[CATAT] %s ke %s: %s",
                teksSimpan, FileBersih, saved and "berhasil" or "GAGAL menulis file"))
            if saved and LoadWorldSet(FileBersih)[target] ~= nil then
                SweepDirtfarm()
            end
            AddHistory(string.format("Bot %s: Lolos inspeksi final", tostring(GetMyIndex())))
        else
            fails[target] = (fails[target] or 0) + 1
            AddHistory(string.format("Bot %s: gagal bersihkan world (%d/%d)",
                tostring(GetMyIndex()), fails[target], MaxGagal))
            if fails[target] >= MaxGagal then
                fails[target] = 0
                cycles[target] = (cycles[target] or 0) + 1
                if cycles[target] >= MaxSiklusGagal then
                    -- gagal terus berulang kali: buang supaya bot tidak macet
                    dropped[target] = ""
                    myWorlds[target], cycles[target] = nil, nil
                    DropWorld(target, "gagal berulang")
                    AddHistory(string.format("Bot %s: world dibuang (gagal berulang)",
                        tostring(GetMyIndex())))
                else
                    retryAt[target] = os.time() + JedaUlangDetik
                end
            end
        end

        SendWebhook()
        if not RestockAtStorage(storageName) then
            SendWebhook("Item tas belum sesuai kuota")
        end
    end

    while true do
        local ok, err = pcall(Step)
        if not ok then
            -- error sekarang dicatat, tidak lagi hilang tanpa jejak
            print("[ERROR] Step: " .. tostring(err))
            pcall(function() AddHistory("Error: " .. tostring(err)) end)
            ExpectedWorld = ""
            pcall(function() bot.auto_collect = false end)
            if not BuildBlocked then
                SetBuild(build, true)
            end
            Sleep(5000, 8000)
        end
    end
end

-- PERBAIKAN: jika Main crash (mis. error saat leveling), script
-- otomatis dimulai ulang, tidak berhenti diam-diam.
while true do
    local ok, err = pcall(Main)
    if not ok then
        print("[FATAL] " .. tostring(err))
        pcall(function() AddHistory("Fatal: " .. tostring(err)) end)
    else
        print("[INFO] Main berhenti, memulai ulang...")
    end
    ExpectedWorld = ""
    Sleep(10000, 15000)
end
